  \cventry
    {fusion-metrics -- Bộ chỉ số đánh giá chất lượng dung hợp ảnh, trung thành với MATLAB (Tác giả \& người duy trì, 2026)}
    {}
    {
      \begin{cvitems}
        \item {Thư viện Python gồm 24 chỉ số đánh giá dung hợp ảnh (Q\textsuperscript{AB/F}, MI, SSIM, SD, NCIE, FMI, FSIM, VIF/VIFF, SCD, PSNR, CC, SF, entropy, average gradient), dịch từng dòng từ các bản cài đặt MATLAB tham chiếu.}
        \item {Kiểm chứng từng giá trị với file .m gốc: độ lệch ở mức sai số làm tròn ($\leq$ 1e-12) trên mọi chỉ số, kể cả các chi tiết riêng của MATLAB (căn tâm conv2 'same', bão hoà uint8, làm mảnh cạnh của edge(), thứ tự cộng).}
        \item {Phát hành trên PyPI (\texttt{pip install fusion-metrics}), có công cụ dòng lệnh để chấm cả một phương pháp, CI trên Python 3.10--3.13 và DOI Zenodo (10.5281/zenodo.24536175).}
        \item {\href{https://github.com/VuVietAnh2309/fusion-metrics}{github.com/VuVietAnh2309/fusion-metrics} · \href{https://pypi.org/project/fusion-metrics/}{pypi.org/project/fusion-metrics}}
      \end{cvitems}
    }

\end{cventries}
